on_actions = {
	on_startup = {
		effect = {
			GER = {
				if = {
					limit = { is_ai = no }
					set_temp_variable = { codex_origin = 1 }
					codex_emit_state = yes
				}
			}
		}
	}
	on_daily_GER = {
		effect = {
			if = {
				limit = { is_ai = no }
				set_temp_variable = { codex_origin = 0 }
				codex_emit_state = yes
			}
		}
	}
}
